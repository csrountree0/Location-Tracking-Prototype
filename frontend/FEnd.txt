import React, { useState, useEffect } from 'react';
import { MapContainer, TileLayer, Marker, Popup, useMap } from 'react-leaflet';
import { Battery, Navigation, Search, MapPin, AlertTriangle, Zap, Menu } from 'lucide-react';
import 'leaflet/dist/leaflet.css';
import L from 'leaflet';

// --- Configuration & Mock Data ---

// Fix for default Leaflet marker icons in React
delete L.Icon.Default.prototype._getIconUrl;
L.Icon.Default.mergeOptions({
  iconRetinaUrl: 'https://cdnjs.cloudflare.com/ajax/libs/leaflet/1.7.1/images/marker-icon-2x.png',
  iconUrl: 'https://cdnjs.cloudflare.com/ajax/libs/leaflet/1.7.1/images/marker-icon.png',
  shadowUrl: 'https://cdnjs.cloudflare.com/ajax/libs/leaflet/1.7.1/images/marker-shadow.png',
});

// Mock Golf Course Center (approximate lat/long)
const COURSE_CENTER = [27.9506, -82.4572]; // Example location

// Mock Cart Data
const INITIAL_CARTS = [
  { id: 'C-101', lat: 27.9510, lng: -82.4580, status: 'Active', battery: 85, rider: 'Guest', speed: 12 },
  { id: 'C-102', lat: 27.9525, lng: -82.4560, status: 'Idle', battery: 45, rider: 'Staff', speed: 0 },
  { id: 'C-103', lat: 27.9490, lng: -82.4590, status: 'Active', battery: 12, rider: 'Guest', speed: 8 }, // Low battery
  { id: 'C-104', lat: 27.9500, lng: -82.4550, status: 'Maintenance', battery: 0, rider: 'None', speed: 0 },
  { id: 'C-105', lat: 27.9515, lng: -82.4565, status: 'Active', battery: 92, rider: 'Guest', speed: 14 },
];

// --- Sub-Components ---

// Helper to pan map when a cart is selected from the sidebar
const MapUpdater = ({ center }) => {
  const map = useMap();
  useEffect(() => {
    map.flyTo(center, 16);
  }, [center, map]);
  return null;
};

const StatusBadge = ({ status }) => {
  const colors = {
    Active: 'bg-green-100 text-green-800 border-green-200',
    Idle: 'bg-gray-100 text-gray-800 border-gray-200',
    Maintenance: 'bg-red-100 text-red-800 border-red-200',
  };
  return (
    <span className={`px-2 py-0.5 rounded-full text-xs font-medium border ${colors[status] || colors.Idle}`}>
      {status}
    </span>
  );
};

const BatteryIndicator = ({ level }) => {
  let color = 'text-green-600';
  if (level < 50) color = 'text-yellow-500';
  if (level < 20) color = 'text-red-500';

  return (
    <div className="flex items-center gap-1">
      <div className={`relative flex items-center justify-center ${color}`}>
        <Battery size={18} />
        <span className="absolute text-[8px] font-bold text-current">{level}</span>
      </div>
    </div>
  );
};

// --- Main Dashboard Component ---

export default function GolfCartDashboard() {
  const [carts, setCarts] = useState(INITIAL_CARTS);
  const [selectedCartId, setSelectedCartId] = useState(null);
  const [filter, setFilter] = useState('');
  const [sidebarOpen, setSidebarOpen] = useState(true);

  // Derived state
  const selectedCart = carts.find(c => c.id === selectedCartId);
  const filteredCarts = carts.filter(c => 
    c.id.toLowerCase().includes(filter.toLowerCase()) || 
    c.status.toLowerCase().includes(filter.toLowerCase())
  );
  
  // Stats
  const activeCount = carts.filter(c => c.status === 'Active').length;
  const lowBatteryCount = carts.filter(c => c.battery < 20).length;

  return (
    <div className="flex h-screen w-full bg-gray-50 overflow-hidden font-sans text-slate-800">
      
      {/* Sidebar List */}
      <aside className={`flex-shrink-0 bg-white border-r border-gray-200 transition-all duration-300 flex flex-col ${sidebarOpen ? 'w-80' : 'w-0'}`}>
        <div className="p-4 border-b border-gray-100">
          <div className="flex items-center justify-between mb-4">
            <h1 className="font-bold text-xl flex items-center gap-2 text-emerald-800">
              <Zap size={24} className="fill-emerald-500 text-emerald-600"/>
              CartTracker
            </h1>
          </div>
          
          {/* Stats Row */}
          <div className="grid grid-cols-2 gap-2 mb-4">
            <div className="bg-emerald-50 p-3 rounded-lg border border-emerald-100">
              <div className="text-xs text-emerald-600 font-semibold uppercase">Active</div>
              <div className="text-2xl font-bold text-emerald-800">{activeCount}</div>
            </div>
            <div className="bg-orange-50 p-3 rounded-lg border border-orange-100">
              <div className="text-xs text-orange-600 font-semibold uppercase">Low Batt</div>
              <div className="text-2xl font-bold text-orange-800">{lowBatteryCount}</div>
            </div>
          </div>

          {/* Search */}
          <div className="relative">
            <Search className="absolute left-3 top-2.5 text-gray-400" size={16} />
            <input 
              type="text" 
              placeholder="Search ID or status..." 
              className="w-full pl-9 pr-4 py-2 bg-gray-50 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500"
              value={filter}
              onChange={(e) => setFilter(e.target.value)}
            />
          </div>
        </div>

        {/* Scrollable Cart List */}
        <div className="flex-1 overflow-y-auto">
          {filteredCarts.map(cart => (
            <div 
              key={cart.id}
              onClick={() => setSelectedCartId(cart.id)}
              className={`p-4 border-b border-gray-100 cursor-pointer hover:bg-gray-50 transition-colors ${selectedCartId === cart.id ? 'bg-emerald-50 border-emerald-200' : ''}`}
            >
              <div className="flex justify-between items-start mb-1">
                <span className="font-semibold text-gray-900">{cart.id}</span>
                <StatusBadge status={cart.status} />
              </div>
              <div className="flex justify-between items-center text-sm text-gray-500 mt-2">
                <div className="flex items-center gap-4">
                  <BatteryIndicator level={cart.battery} />
                  {cart.status === 'Active' && <span className="text-xs">{cart.speed} mph</span>}
                </div>
                <div className="flex items-center gap-1 text-xs">
                  <Navigation size={12} />
                  <span>Locate</span>
                </div>
              </div>
            </div>
          ))}
        </div>
      </aside>

      {/* Main Content (Map) */}
      <main className="flex-1 flex flex-col relative">
        {/* Mobile/Toggle Header */}
        <header className="h-16 bg-white border-b border-gray-200 flex items-center px-4 justify-between lg:hidden z-10">
           <button onClick={() => setSidebarOpen(!sidebarOpen)} className="p-2 hover:bg-gray-100 rounded">
             <Menu size={20} />
           </button>
           <span className="font-bold">Dashboard</span>
        </header>

        {/* Floating Toggle for Desktop */}
        <button 
          onClick={() => setSidebarOpen(!sidebarOpen)} 
          className="absolute top-4 left-4 z-[400] bg-white p-2 rounded shadow-md border border-gray-200 hidden lg:block hover:bg-gray-50"
        >
          <Menu size={20} />
        </button>

        <div className="flex-1 relative z-0">
          <MapContainer center={COURSE_CENTER} zoom={15} style={{ height: '100%', width: '100%' }}>
            <TileLayer
              attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
              url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
            />
            
            {/* Render Carts on Map */}
            {carts.map(cart => (
              <Marker 
                key={cart.id} 
                position={[cart.lat, cart.lng]}
                eventHandlers={{
                  click: () => setSelectedCartId(cart.id),
                }}
              >
                <Popup>
                  <div className="min-w-[150px]">
                    <h3 className="font-bold text-lg mb-1">{cart.id}</h3>
                    <div className="space-y-1 text-sm">
                      <p className="flex justify-between">Status: <strong className={cart.status === 'Active' ? 'text-green-600' : 'text-gray-600'}>{cart.status}</strong></p>
                      <p className="flex justify-between">Battery: <strong>{cart.battery}%</strong></p>
                      <p className="flex justify-between">Rider: <strong>{cart.rider}</strong></p>
                    </div>
                  </div>
                </Popup>
              </Marker>
            ))}

            {/* Programmatically move map if a cart is selected from the list */}
            {selectedCart && <MapUpdater center={[selectedCart.lat, selectedCart.lng]} />}
          </MapContainer>
        </div>
        
        {/* Bottom Overlay Info (Optional Detail View) */}
        {selectedCart && (
          <div className="absolute bottom-6 left-1/2 transform -translate-x-1/2 bg-white rounded-xl shadow-xl p-4 border border-gray-200 w-[90%] max-w-lg z-[400] flex justify-between items-center">
             <div>
                <p className="text-xs text-gray-500 uppercase font-bold tracking-wide">Selected Unit</p>
                <h2 className="text-xl font-bold text-gray-800">{selectedCart.id}</h2>
             </div>
             <div className="flex gap-6">
                <div className="text-center">
                   <p className="text-xs text-gray-400">Battery</p>
                   <p className={`font-bold ${selectedCart.battery < 20 ? 'text-red-600' : 'text-green-600'}`}>{selectedCart.battery}%</p>
                </div>
                <div className="text-center">
                   <p className="text-xs text-gray-400">Speed</p>
                   <p className="font-bold text-gray-800">{selectedCart.speed} mph</p>
                </div>
                <div className="text-center">
                   <p className="text-xs text-gray-400">Rider</p>
                   <p className="font-bold text-gray-800">{selectedCart.rider}</p>
                </div>
             </div>
          </div>
        )}
      </main>
    </div>
  );
}
